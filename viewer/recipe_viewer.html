CLI 사용법

# Document JSON -> Vector JSON (명령어 추출)
node tools/recipe/vector_converter.mjs \
  --to-vector \
  --doc output.drawtool.json \
  --out output.vector.json

# Vector JSON -> Document JSON (복원)
node tools/recipe/vector_converter.mjs \
  --to-doc \
  --vector output.vector.json \
  --out restored.drawtool.json

# 전체 파이프라인
node tools/recipe/recipe_converter.mjs \
  --to-doc \
  --recipe legacy.json \
  --out temp.drawtool.json

node tools/recipe/vector_converter.mjs \
  --to-vector \
  --doc temp.drawtool.json \
  --out final.vector.json
